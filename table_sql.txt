-- Table user : données de connexion et rôles
CREATE TABLE user (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    username TEXT NOT NULL,
    password_hash TEXT NOT NULL,
    role TEXT NOT NULL DEFAULT 'user' CHECK (role IN ('user', 'staff', 'admin')),
    banned INTEGER NOT NULL DEFAULT 0 CHECK (banned IN (0,1))
);

-- Table place : lieux du campus
CREATE TABLE place (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL
);

-- Table affluence : signalements des utilisateurs
CREATE TABLE affluence (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id INTEGER NOT NULL,
    place_id INTEGER NOT NULL,
    niveau TEXT NOT NULL CHECK (niveau IN ('vide', 'faible', 'moyen', 'fort', 'plein')),
    timestamp TIMESTAMP NOT NULL,
    FOREIGN KEY (user_id) REFERENCES user(id),
    FOREIGN KEY (place_id) REFERENCES place(id)
);

-- Table computer_bu : ordinateurs de la bibliothèque universitaire
CREATE TABLE computer_bu (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    name TEXT NOT NULL,
    user_id INTEGER,
    FOREIGN KEY (user_id) REFERENCES user(id)
);

-- Table menu_ru : contenu du menu du restaurant universitaire
CREATE TABLE menu_ru (
    id INTEGER PRIMARY KEY CHECK (id = 1),
    content TEXT NOT NULL,
    updated_at TIMESTAMP NOT NULL,
    updated_by INTEGER,
    FOREIGN KEY (updated_by) REFERENCES user(id)
);
